\begin{frame}{Das Problem: die Klickrate von Mullers Ratsche}
  \small
  Unendliches SDE-System mit Fleming--Viot-Rauschen, $k=0,1,2,\dots$, $\sum_k X_k=1$:
  \[
    dX_k = \Big(\alpha\Big(\sum_j (j-k)X_j\Big)X_k + \lambda(X_{k-1}-X_k)\Big)dt
      + \sum_{\ell\neq k}\sqrt{\tfrac1\nu X_kX_\ell}\,dW_{k\ell}.
  \]
  $X_k$ ist der Anteil mit $k$ schädlichen Mutationen. $K^\ast(t)=\inf\{k: X_k(t)>0\}$ springt
  gelegentlich um eins: ein \textbf{Klick}.
  \begin{itemize}
  \item Gesucht: die Rate $\mathfrak r=\lim_t K^\ast(t)/t$. Wegen
    $\mathfrak r(\alpha,\lambda,\nu)=\alpha\,\mathfrak r(1,\theta,\nu\alpha)$ zwei Parameter:
    $\theta=\lambda/\alpha$ und $n=\nu\alpha$.
  \item Für $n\to\infty$ ist das \textbf{kleines Rauschen}. Der Fixpunkt ist
    $\pi=\mathrm{Poi}(\theta)$, ein Klick ist ein seltener Austritt. Vermutung:
    $\tfrac1n\log\mathfrak r\to-V_\ast(\theta)$, mit $V_\ast$ dem Quasipotential.
  \item Die Literatur schließt auf eine Gleichung für $X_0$ allein:
    $V_{1d}(\theta)=2\theta\big[(1+\theta)\log(1+1/\theta)-1\big]$.
    Der Fehler geht bis zum Faktor~2.
  \end{itemize}
  \begin{block}{Der Auftrag}
    Über die 1d-Heuristik hinauskommen. Eine erneute 1d-Schließung ist \textbf{kein}
    Fortschritt.
  \end{block}
\end{frame}

\begin{frame}{Die Idee der Läufe (1.--3.\,8.\,2026)}
  \begin{itemize}
  \item \textbf{Ich bin im Urlaub.} Ein Rechner im Büro rechnet eine Woche lang
    allein, alle vier Stunden ein Lauf von \texttt{claude -p}.
  \item \textbf{Kein Lauf hat ein Gedächtnis.} Der Stand steht nur in
    \datei{RESEARCH\_LOG.md} und in den Commits. Jeder Lauf liest das Log vollständig
    und hängt einen Eintrag an.
  \item \textbf{Der Name ist Programm.} Eine Ratsche dreht nur in eine Richtung: Nichts
    Verifiziertes wird wieder aufgegeben.
  \end{itemize}
  \medskip
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Ein Log-Eintrag}
        \footnotesize
        \textbf{Gelesen}: der Stand\\
        \textbf{Versucht}: der Ansatz\\
        \textbf{Ergebnis}: BEWIESEN / VERMUTUNG / SACKGASSE\\
        \textbf{Nächster Schritt}: ein konkreter Vorschlag
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \footnotesize
      \begin{itemize}
      \item \emph{Ehrlichkeit vor Fortschritt.}
      \item Sackgassen sind Ergebnisse.
      \item Erfinde keine Literatur.
      \item Versuche keine Sackgasse erneut ohne neuen Grund.
      \item Keine Rückfragen: Entscheide selbst und dokumentiere es.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Fehlschritt 1: der Lauf hätte nie angefangen}
  Der Probelauf am 3.\,8. vormittags fand \textbf{vier Fehler}, jeder davon hätte den
  unbeaufsichtigten Lauf sofort gekippt:
  \begin{itemize}
  \item Das Repo lag nicht unter dem Pfad, den das Skript kannte:

    sofort ab.
  \item Die GPU-Simulation lag woanders als angegeben.
  \item Eine unnötige \texttt{venv}-Anweisung.
  \end{itemize}
  \medskip
  \textbf{Inhaltlich wichtiger:} In \datei{Papers/ratchet/} lag weit mehr, als
  \datei{PROBLEM.md} wusste, darunter \datei{ldp.tex} mit der schon hergeleiteten
  Hamiltonfunktion. Ohne Eintrag hätte der Lauf das nachgebaut. Seitdem steht jede Datei
  mit der geprüften Zuordnung \enquote{Datei $\mapsto$ Arbeit} in der Problemstellung.
\end{frame}

\begin{frame}{Fehlschritt 2: eine Numerik-Maschine}
  \small
  \begin{columns}[T]

      \textbf{So war es angelegt}
      \begin{itemize}
      \item \datei{PROBLEM.md}: die Simulation als \enquote{Verifier (verbindlich)}
      \item Prompt: \enquote{Prüfe numerisch}
      \item drei Ergebnistypen gleichrangig
      \end{itemize}
      \textbf{Iteration 1} lieferte vier gute Messergebnisse, keinen Beweisversuch, und
      schlug wieder etwas Numerisches vor.
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Anmerkung des Betreuers}, 3.\,8., von Hand ins Log, \emph{keine}
      Iteration:
      \begin{itemize}
      \item \textbf{Ziel ist ein Satz, keine Tabelle.}
      \item Die Simulation ist \textbf{Sanity-Check}: Sie kann widerlegen, nie
        bestätigen. Numerisch Gestütztes bleibt VERMUTUNG.
      \item $V_\ast$ ist ein Infimum, Schranken brauchen \textbf{kein} LDP:
      \end{itemize}
      \footnotesize
      \textbf{B1}: jeder Pfad $\varphi$ gibt $V_\ast\le I(\varphi)$.\\
      \textbf{B2}: jede Subsolution $F$ gibt $V_\ast\ge\inf\{F: x_0=0\}$.
    \end{column}
  \end{columns}
  \bigskip
  Iteration 2 bewies darauf einen Negativsatz: Die $(x_0,m_1)$-Schließung liefert keine
  untere Schranke.
\end{frame}

\begin{frame}{Fehlschritte in der Mathematik}
  \small
  \textbf{Widerlegt und dokumentiert: das ist der Normalfall}
  \begin{itemize}
  \item Der Entropie-Ansatz für B2 (It.~6): Er bricht an der Schwelle $\theta_c=2.4088$.
  \item (R59-a) widerlegt und repariert (20.\,8.): das Produkt $x_0m_1$ statt des Paars
    $(x_0,m_1)$.
  \item Die Klausel (K1) widerlegt (It.~72), darauf zwei neue Wege.
  \item B8 (moderate Abweichungen): nach einem Lauf negativ. $c_{MD}\sim1/(2\log\theta)\to0$
    statt eines Werts in $(0,1)$. Die Reihe war auf sechs Läufe geplant.
  \end{itemize}
  \textbf{Gefunden erst beim Gegenlesen von Hand} (22.\,8., fünf Befunde, keine Aussage fällt)
  \begin{itemize}
  \item Die Einleitung behauptete einen vollen Limes. Bewiesen ist er nur unter der
    Hypothese (RET).
  \item Eine Vergleichsfunktion löste die falsche Riccati-Gleichung: Die Differenz ist
    genau $\theta/2$.
  \item Eine Konstante zitiert mit $\delta/4$ statt $\delta/8$. Eine Schranke war
    \emph{kleiner} als die bewiesene.
  \end{itemize}
  Die Läufe prüfen ihre eigenen Beweise, aber schlecht die \textbf{Zitatketten}
  zwischen Iterationen.
\end{frame}

\begin{frame}{Was ich daraus gelernt habe}
  \begin{itemize}
  \setlength\itemsep{6pt}
  \item \textbf{Die Problemstellung ist der Hebel.} Beide großen Fehlschritte, die
    Numerik-Maschine und das nie geöffnete Buch, standen so in \datei{PROBLEM.md}. Die
    Läufe haben genau das getan, was dort stand.
  \item \textbf{Die Richtung bleibt beim Nutzer.} Ein Lauf merkt nicht, dass ein Weg
    ausgereizt ist. Er macht ihn nur feiner.
  \item \textbf{Zuschnitt schlägt Freiheit.} Seit den Reihen gibt es pro Lauf ein benanntes
    Ziel und eine Stoppgrenze, und aus dem LDP wurde in zwei Wochen ein Manuskript.
  \item \textbf{Ein Verifier, der nur widerlegen kann, reicht nicht.} Die Beweise standen
    in Prosa. Was sie trägt, zeigte erst das Gegenlesen.
  \item \textbf{Der Betrieb kostet mehr als gedacht.} Pfade, Kontingente, Stoppzähler und
    Crontab-Reste: Jede dieser Pannen ist einmal passiert und steht seitdem als Kommentar im
    Skript.
  \end{itemize}
\end{frame}
