\begin{frame}{Woher ein Lauf seinen Auftrag bekommt}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \datei{scripts/facts\_prompt.md}, bei jedem Start frisch gelesen:
      \begin{enumerate}
      \item \textbf{Vorrangige Aufgaben}, von mir datiert, mit
        \textbf{Schritten in fester Reihenfolge}.
      \item \textbf{Stehende Regeln}: minimale Voraussetzungen, Übersetzen,
        der Pflichtbericht am Ende.
      \item Erledigte Aufgaben wandern nach \datei{facts\_prompt\_archiv.md}.
        Kein Lauf liest das Archiv.
      \end{enumerate}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Ein Lauf hat kein Gedächtnis}
        Jeder Lauf beginnt leer. Was er wissen muss, steht an zwei Stellen:
        \begin{itemize}
        \item in \datei{Facts/INVENTAR.md}: Berichte, Zuordnungen
          \enquote{Manuskript $\mapsto$ Lean-Name}, Befunde;
        \item in den Commits auf \texttt{facts-inventory}.
        \end{itemize}
      \end{block}
      \begin{block}{Ein Schritt ist ein Lauf und ein benanntes Ziel}
        Jeder Lauf nennt am Ende das Ziel für den nächsten, und das ist immer der
        nächste offene Schritt \emph{meiner} Liste.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Was ein Lauf darf und was nicht}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textcolor{darkgreen}{\textbf{Darf}}
      \begin{itemize}
      \item Lean schreiben und übersetzen (\texttt{lake}, \texttt{lean})
      \item \datei{check\_master.py}: baut alle vier Dateien gegen Mathlib
        \texttt{master}. Diese Prüfung zählt.
      \item Mathlib und BrownianMotion \textbf{lesen}
      \item Befunde ins Inventar schreiben, mit Zeilennummer
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textcolor{darkred}{\textbf{Darf nicht}}
      \begin{itemize}
      \item das Manuskript ändern
      \item eine README anfassen
      \item \textbf{nach außen} gehen: kein PR, kein Issue, kein Kommentar
      \item Prozesse im Hintergrund starten
      \item Dateien löschen (\texttt{rm} ist nicht freigegeben)
      \end{itemize}
    \end{column}
  \end{columns}
  \bigskip
  \small Die Rechte hängen am \textbf{Namen des Befehls}. Nicht Freigegebenes wird im
  Modus \texttt{-p} verweigert, nicht erfragt. Ein Lauf bleibt also nie hängen und
  wartet auf mich.
\end{frame}

\begin{frame}{Mein Einfluss}
  \begin{enumerate}
  \setlength\itemsep{4pt}
  \item \textbf{Aufträge stellen.} In einer interaktiven Sitzung formuliert und
    nach \texttt{master} gepusht. Ich setze die Reihenfolge; Läufe wählen sich
    kein eigenes Ziel.
  \item \textbf{Anhalten und freigeben.} Mit \texttt{NOT\_BEFORE} in
    \datei{facts\_gate.sh}, zum Beispiel am 15.\,9. (Kontingent erschöpft) und am
    22.\,9. (keine Läufe mehr, ohne Enddatum), wieder frei am 24.\,9.
  \item \textbf{Das Manuskript korrigieren.} Die Läufe finden Fehler, ich entscheide
    und ändere.
  \item \textbf{Entscheiden, was nach außen geht.} Mathlib-Kandidaten und Lücken bei
    BrownianMotion sammeln sich im Inventar, ich entscheide über sie.
  \item \textbf{Abnehmen.} Der Merge von \texttt{facts-inventory} nach
    \texttt{master}, zuletzt am 22.\,9.
  \item \textbf{Aufräumen und nachsteuern,} wenn Läufe festhängen oder sich irren.
  \end{enumerate}
\end{frame}

\begin{frame}{Was bleibt beim Nutzer}
  \begin{itemize}
  \setlength\itemsep{6pt}
  \item \textbf{Die mathematische Richtigkeit der Aussagen.} \datei{check\_master.py}
    und \texttt{\#print axioms} prüfen die Beweise, nicht, ob die Lean-Aussage das
    Manuskript trifft.
  \item \textbf{Die Richtung.} Läufe arbeiten ab, was dasteht, und sie sind gut
    darin. Was als Nächstes wichtig ist, sagen sie nicht.
  \item \textbf{Die Außenwirkung.} Mathlib und BrownianMotion bekommen nur, was ich
    freigebe.
  \item \textbf{Die Lesbarkeit.} Die Läufe schreiben viel. Das Inventar hat mehr als
    65\,000 Zeilen, und die READMEs redigiere ich selbst.
  \end{itemize}
\end{frame}
